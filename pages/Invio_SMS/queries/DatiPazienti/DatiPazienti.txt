SELECT TOP 100
    PAPATIENTID, 
    PASURNAME AS Cognome,
    PANAME AS Nome,
    PAFISCALCODE AS [Codice Fiscale],
    PAPHONE1 AS [Telefono 1],
    PAPHONE2 AS [Telefono 2]
FROM 
    ba_patient WITH (NOLOCK)
WHERE 
    -- Eliminati i controlli IS NOT NULL ridondanti (gestiti già dal LIKE se i campi di input hanno testo)
    (PAVALIDTO IS NULL OR PAVALIDTO > GETDATE())
    
    -- Filtri di ricerca ottimizzati per evitare scansioni inutili se l'input è vuoto
    AND (
        '{{ txtCognome.text }}' = '' 
        OR PASURNAME LIKE '{{ txtCognome.text }}' + '%' -- Rimosso il primo '%' se cerchi per inizio stringa
    )
    AND (
        '{{ txtNome.text }}' = '' 
        OR PANAME LIKE '{{ txtNome.text }}' + '%'
    )
ORDER BY 
    PASURNAME, PANAME;
