SELECT
    O.MVCODCON                    AS CodiceFornitore,
    C.ANDESCRI                    AS Fornitore,
    COUNT(DISTINCT O.MVSERIAL)    AS NumeroOrdini,
    SUM(D.MVVALRIG)               AS TotaleAcquistato
FROM A0001DOC_MAST O
LEFT JOIN A0001CONTI C
    ON C.ANCODICE = O.MVCODCON
INNER JOIN A0001DOC_DETT D
    ON D.MVSERIAL = O.MVSERIAL
WHERE
    CONVERT(DATE, O.MVDATREG) >= CONVERT(DATE, CAST('{{ dat_inizio.formattedDate }}' AS DATETIME))
    AND CONVERT(DATE, O.MVDATREG) <= CONVERT(DATE, CAST('{{ dat_fine.formattedDate }}' AS DATETIME))
    AND O.MVTIPDOC = 'ORDFO'
    AND D.MVCODART IS NOT NULL
    AND D.MVCODART <> ''
    AND D.MVCODART NOT LIKE '%.%'
    AND D.MVCODART NOT LIKE ',%'
    AND D.MVQTAMOV > 0
GROUP BY
    O.MVCODCON,
    C.ANDESCRI
ORDER BY
    C.ANDESCRI;
