SELECT
    O.MVCODCON                          AS CodiceFornitore,
    C.ANDESCRI                          AS Fornitore,
    A.ARCATCON                          AS CodiceCategoria,
    CC.C1DESCRI                         AS Categoria,
    F.FACODICE                          AS CodiceFamiglia,
    F.FADESCRI                          AS FamigliaAdHoc,
    BAF.FADESCRI                        AS FamigliaH2O,
    -- Anno corrente
    COUNT(DISTINCT CASE WHEN YEAR(O.MVDATDOC) = YEAR(GETDATE()) THEN O.MVSERIAL END) AS OrdiniAnnoCorrente,
    CAST(SUM(CASE WHEN YEAR(O.MVDATDOC) = YEAR(GETDATE()) THEN D.MVVALRIG ELSE 0 END) AS DECIMAL(18,2)) AS TotaleAnnoCorrente,
    -- Anno scorso
    COUNT(DISTINCT CASE WHEN YEAR(O.MVDATDOC) = YEAR(GETDATE()) - 1 THEN O.MVSERIAL END) AS OrdiniAnnoScorso,
    CAST(SUM(CASE WHEN YEAR(O.MVDATDOC) = YEAR(GETDATE()) - 1 THEN D.MVVALRIG ELSE 0 END) AS DECIMAL(18,2)) AS TotaleAnnoScorso,
    -- Due anni fa
    COUNT(DISTINCT CASE WHEN YEAR(O.MVDATDOC) = YEAR(GETDATE()) - 2 THEN O.MVSERIAL END) AS OrdiniDueAnniFa,
    CAST(SUM(CASE WHEN YEAR(O.MVDATDOC) = YEAR(GETDATE()) - 2 THEN D.MVVALRIG ELSE 0 END) AS DECIMAL(18,2)) AS TotaleDueAnniFa
FROM A0001DOC_MAST O
LEFT JOIN A0001CONTI C
    ON C.ANCODICE = O.MVCODCON
INNER JOIN A0001DOC_DETT D
    ON D.MVSERIAL = O.MVSERIAL
LEFT JOIN A0001ART_ICOL A
    ON A.ARCODART = D.MVCODART
LEFT JOIN A0001CACOARTI CC
    ON CC.C1CODICE = A.ARCATCON
LEFT JOIN A0001FAM_ARTI F
    ON F.FACODICE = A.ARCODFAM
LEFT JOIN [H2O_MD]..ba_art_family BAF
    ON BAF.FAIDFAMYLI = F.FACODICE
WHERE
    O.MVTIPDOC = 'ORDFO'
    AND O.MVCODCON = '{{ fornitoriMD.selectedOptionValue }}'
    AND YEAR(O.MVDATDOC) >= YEAR(GETDATE()) - 2
    AND D.MVCODART IS NOT NULL
    AND D.MVCODART <> ''
    AND D.MVCODART NOT LIKE '%.%'
    AND D.MVCODART NOT LIKE ',%'
    AND D.MVQTAMOV > 0
GROUP BY
    O.MVCODCON,
    C.ANDESCRI,
    A.ARCATCON,
    CC.C1DESCRI,
    F.FACODICE,
    F.FADESCRI,
    BAF.FADESCRI
ORDER BY
    C.ANDESCRI,
    CC.C1DESCRI,
    F.FADESCRI;
