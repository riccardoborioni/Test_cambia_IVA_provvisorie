SELECT
  h.EVMHYEAR,
  h.EVARTICLEDESCRI,
  h.EVWARDEFFECTIVEID,
  h.EVEVENTDATEBEGIN,
  h.EVCREATEDBY,
  h.EVINSTANCEID,
  u.[name] AS EVCREATEDBY_NAME,
  hos.HORESERVATIONCOUNTER,
  hos.HOLASTNAME,
  hos.HOFIRSTNAME
FROM ba_hosp_administration h WITH (NOLOCK)
INNER JOIN ba_articles a WITH (NOLOCK)
  ON a.ARCODART = h.EVARTICLEID
INNER JOIN ba_hospitalization hos WITH (NOLOCK)
  ON hos.HOHOSPID = h.EVHOSPID
LEFT JOIN cpusers u WITH (NOLOCK)
  ON u.[code] = h.EVCREATEDBY
WHERE a.ARCODART = {{ Select1.selectedOptionValue }}
  AND h.EVMHYEAR = {{ Number(InpAnno.text) || 2026 }}
ORDER BY h.EVMHYEAR DESC;
